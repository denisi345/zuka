<script>
const msgs=document.getElementById('msgs');
const input=document.getElementById('msgInput');
function sendMsg(){
  if(!input.value.trim())return;
  const div=document.createElement('div');
  div.className='bubble me'; div.textContent=input.value;
  msgs.appendChild(div); input.value=''; msgs.scrollTop=msgs.scrollHeight;
  localStorage.setItem('zuka_chat',msgs.innerHTML);
  setTimeout(()=>{ const r=document.createElement('div'); r.className='bubble them'; r.textContent='Got it! 👊'; msgs.appendChild(r); msgs.scrollTop=msgs.scrollHeight; },700);
}
input.addEventListener('keydown',e=>{if(e.key==='Enter')sendMsg()});
function addStory(){
  const url=prompt('Paste image URL for your story:','https://i.pravatar.cc/150');
  if(!url)return;
  const s=document.createElement('div'); s.className='story';
  s.innerHTML=`<div class="story-ring"><img src="${url}"></div><p>You</p>`;
  document.getElementById('stories').appendChild(s);
}
async function startVideo(){
  document.getElementById('videoModal').style.display='flex';
  try{ const stream=await navigator.mediaDevices.getUserMedia({video:true,audio:true}); document.getElementById('localVideo').srcObject=stream; window._stream=stream; }catch(e){alert('Please allow camera access');}
}
function stopVideo(){
  document.getElementById('videoModal').style.display='none';
  if(window._stream)window._stream.getTracks().forEach(t=>t.stop());
}
msgs.innerHTML=localStorage.getItem('zuka_chat')||msgs.innerHTML;
</script>
